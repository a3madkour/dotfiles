** Exec path from shell
GUI Emacs launched from the dock/Finder does not inherit the login shell
=$PATH=, so binaries installed under =~/.local/bin= (e.g. =uv tool install=d
CLIs like [[https://github.com/a3madkour/org-math-lint][org-math-lint]]) are invisible. Copy =PATH= (and friends) from the
login shell so =exec-path= matches the terminal.
#+begin_src emacs-lisp

** Smart-parens
Adapted from [[https://github.com/doomemacs/doomemacs/blob/master/modules/config/default/config.el][Doom Emacs's smartparens config]] (=modules/config/default/config.el=); the =(dolist (brace ...))= block and =:unless= rationale are verbatim.
#+begin_src emacs-lisp
(use-package smartparens
  :diminish smartparens-mode
  :defer 1
  :config
  ;; Load default smartparens rules for various languages
  (require 'smartparens-config)
  (setq sp-max-prefix-length 25)
  (setq sp-max-pair-length 4)
  (setq sp-highlight-pair-overlay nil
		sp-highlight-wrap-overlay nil
		sp-highlight-wrap-tag-overlay nil)

  (with-eval-after-load 'evil
	(setq sp-show-pair-from-inside t)
	(setq sp-cancel-autoskip-on-backward-movement nil)
	(setq sp-pair-overlay-keymap (make-sparse-keymap)))

  (let ((unless-list '(sp-point-before-word-p
					   sp-point-after-word-p
					   sp-point-before-same-p)))
	(sp-pair "'"  nil :unless unless-list)
	(sp-pair "\"" nil :unless unless-list))

  ;; In lisps ( should open a new form if before another parenthesis
  (sp-local-pair sp-lisp-modes "(" ")" :unless '(:rem sp-point-before-same-p))

  ;; Don't do square-bracket space-expansion where it doesn't make sense to
  (sp-local-pair '(emacs-lisp-mode org-mode markdown-mode gfm-mode)
				 "[" nil :post-handlers '(:rem ("| " "SPC")))


  (dolist (brace '("(" "{" "["))
	(sp-pair brace nil
			 :post-handlers '(("||\n[i]" "RET") ("| " "SPC"))
			 ;; Don't autopair opening braces if before a word character or
			 ;; other opening brace. The rationale: it interferes with manual
			 ;; balancing of braces, and is odd form to have s-exps with no
			 ;; whitespace in between, e.g. ()()(). Insert whitespace if
			 ;; genuinely want to start a new form in the middle of a word.
			 :unless '(sp-point-before-word-p sp-point-before-same-p)))
  (smartparens-global-mode t))

  "Filter 'recentf-list' to show only files within the current project"
  (interactive)

  		 (root (project-root proj))

(defun uv-activate ()
  "Activate Python environment managed by uv based on current project directory.
Looks for .venv directory in project root and activates the Python interpreter."
  (interactive)
  (let* ((project-root (project-root (project-current t)))
         (venv-path (expand-file-name ".venv" project-root))
         (python-path (expand-file-name
                       (if (eq system-type 'windows-nt)
                           "Scripts/python.exe"
                         "bin/python")
                       venv-path)))
    (if (file-exists-p python-path)
        (progn
          ;; Set Python interpreter path
          (setq python-shell-interpreter python-path)

          ;; Update exec-path to include the venv's bin directory
          (let ((venv-bin-dir (file-name-directory python-path)))
            (setq exec-path (cons venv-bin-dir
                                  (remove venv-bin-dir exec-path))))

          ;; Update PATH environment variable
          (setenv "PATH" (concat (file-name-directory python-path)
                                 path-separator
                                 (getenv "PATH")))

          ;; Update VIRTUAL_ENV environment variable
          (setenv "VIRTUAL_ENV" venv-path)

          ;; Remove PYTHONHOME if it exists
          (setenv "PYTHONHOME" nil)

          (message "Activated UV Python environment at %s" venv-path))
      (error "No UV Python environment found in %s" project-root))))

The =org-cite-*-processor= setup and =citar-embark= activation follow the [[https://github.com/emacs-citar/citar][citar README]] by Bruce D'Arcus; the =a3madkour/citar-org-format-note=, =a3madkour/citar-add-to-reading-queue=, and =a3madkour/supernote-*= helpers are personal.
#+begin_src emacs-lisp

(defun a3madkour/citar-org-format-note(key entry)
  "Format a note from KEY and ENTRY."
  (let* ((template (citar--get-template 'note))
		 (note-meta (when template
					  (citar-format--entry template entry)))
		 (template-path (citar--get-template 'note-file))
		 (note-path (when template-path
					  (citar-format--entry template-path entry)))
		 (note-path-clean (when note-path
							(car (seq-filter (lambda (p) (string-match-p "\\.pdf$" (string-trim p)))
											 (split-string note-path ";")))))
		 (note-tags (cdr (citar-get-field-with-value '("keywords") key)))
		 (formatted-tags (when note-tags
						   (mapconcat (lambda (tag) (concat ":" (replace-regexp-in-string "\\s-+" "" (string-trim tag))))
									  (split-string note-tags "[,;]+")
									  "")))
		 (filepath (expand-file-name
					(concat key ".org")
					(car citar-notes-paths)))
		 (buffer (find-file filepath)))
	(with-current-buffer buffer
	  ;; This just overrides other template insertion.
	  (erase-buffer)
	  (citar-org-roam-make-preamble key)
	  ;; hacky way to insert NOTER_DOCUMENT
	  (goto-char (point-min))
	  (when (search-forward ":END:" nil t)
		(beginning-of-line)
		(let ((doc (and note-path-clean
						(or (a3madkour/normalize-library-path note-path-clean)
							note-path-clean))))
		  (insert ":NOTER_DOCUMENT: ")
		  (when doc (insert doc))
		  (insert "\n")
		  (insert ":ORIGINAL_DOCUMENT: ")
		  (when doc (insert doc))
		  (insert "\n"))
		(insert ":DATE_CREATED: ")
		(insert (format-time-string "[%Y-%m-%d %a]"))
		(insert "\n")
		(insert ":LAST_MODIFIED: ")
		(insert (format-time-string "[%Y-%m-%d %a]"))
		(insert "\n"))	
	  )
	(goto-char (point-max))
	(insert "#+title: ")
	(when template (insert note-meta))
	(insert "\n#+filetags: :Ref")
	(when formatted-tags (insert formatted-tags))
	(insert ":\n")
	(insert "\n\n|\n\n#+print_bibliography:")
	(search-backward "|")
	(delete-char 1)
	(when (fboundp 'evil-insert)
	  (evil-insert 1))))

(defun a3madkour/citar-add-to-reading-queue ()
  "Select one or more citar references and add them to queue.org."
  (interactive)
  (require 'citar)
  (let* ((count (a3madkour/reading-queue-active-count))
         (limit a3madkour/reading-queue-wip-limit))
    (when (>= count limit)
      (user-error "Reading queue WIP limit reached (%d/%d active). Finish something first"
                  count limit))
    (let* ((keys (citar-select-refs))
           (n (length keys))
           (room (- limit count)))
      (when (> n room)
        (user-error "Selected %d ref(s) but only %d slot(s) left under WIP limit (%d/%d active)"
                    n room count limit))
      (with-current-buffer (find-file-noselect (a3madkour/org-file "queue.org"))
        (goto-char (point-max))
        (dolist (key keys)
          (let* ((entry (citar-get-entry key))
                 (title (alist-get "title" entry nil nil #'string=))
                 (author (alist-get "author" entry nil nil #'string=))
                 (year (when-let ((date (alist-get "date" entry nil nil #'string=)))
                         (substring date 0 4)))
                 (display (format "%s (%s)" (or title "Untitled") (or year "")))
                 (text (format "* TODO Read %s :active:toread:\n:PROPERTIES:\n:ENERGY: high\n:CITEKEY: %s\n:AUTHOR: %s\n:END:\n[cite:@%s]\n\n"
                               display key (or author "") key)))
            (insert text)))
        (save-buffer))
      (message "Added %d ref(s) to reading queue (%d/%d active)"
               n (+ count n) limit))))

  (when-let* ((entry (citar-get-entry key))
              (file (alist-get "file" entry nil nil #'string=))

(defun a3madkour/supernote-sync-active-queue ()
  "Stage PDFs for every :active: entry in queue.org into the Supernote outbox.
Files are copied to `a3madkour/supernote-outbox-path'; drag them into
OpenMTP → Document/Inbox to transfer.  Skips files already present
in the outbox.  Shows a summary buffer listing copied, skipped, and
missing items."
  (interactive)
  (require 'citar)
  (let* ((dest-dir (file-name-as-directory
                    (expand-file-name a3madkour/supernote-outbox-path)))
         (keys (a3madkour/supernote--queue-citekeys))
         copied skipped missing)
    (unless (file-directory-p dest-dir)
      (make-directory dest-dir t))
    (dolist (key keys)
      (let ((src (a3madkour/supernote--pdf-for-key key)))
        (cond
         ((or (null src) (not (file-readable-p src)))
          (push key missing))
         (t
          (let ((dest (expand-file-name (file-name-nondirectory src) dest-dir)))
            (if (file-exists-p dest)
                (push (cons key (file-name-nondirectory src)) skipped)
              (copy-file src dest)
              (push (cons key (file-name-nondirectory src)) copied)))))))
    (with-current-buffer (get-buffer-create "*Supernote Sync*")
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert (format "Supernote outbox: %s\n" dest-dir))
        (insert "Drag the staged files into OpenMTP → Document/Inbox.\n\n")
        (insert (format "Copied: %d   Skipped: %d   Missing: %d\n\n"
                        (length copied) (length skipped) (length missing)))
        (when copied
          (insert "Copied:\n")
          (dolist (it (nreverse copied))
            (insert (format "  %s  (%s)\n" (cdr it) (car it))))
          (insert "\n"))
        (when skipped
          (insert "Skipped (already in outbox):\n")
          (dolist (it (nreverse skipped))
            (insert (format "  %s  (%s)\n" (cdr it) (car it))))
          (insert "\n"))
        (when missing
          (insert "Missing PDF:\n")
          (dolist (k (nreverse missing))
            (insert (format "  %s\n" k)))))
      (special-mode))
    (display-buffer "*Supernote Sync*")
    (when (and a3madkour/supernote-reveal-outbox-after-sync copied)
      (pcase system-type
        ('darwin    (call-process "open" nil 0 nil dest-dir))
        ('gnu/linux (call-process "xdg-open" nil 0 nil dest-dir))))
    (message "Supernote outbox: %d copied, %d skipped, %d missing (drag from %s into OpenMTP)"
             (length copied) (length skipped) (length missing) dest-dir)))

(defun a3madkour/normalize-library-path (path)
  "Re-anchor PATH from a bib `file' field onto this machine's home directory.
The bibliography was written on macOS, so entries carry
/Users/a3madkour/Sync/... prefixes that do not resolve elsewhere.  Take
everything from the first `Sync/' path segment onward and expand it under
`~'.  Matching is case-sensitive and requires a whole segment, so a
directory such as `ResilioSync/' is not treated as a match.  Return nil
when PATH is empty or contains no `Sync/' segment."
  (when (and path (stringp path))
    (let ((trimmed (string-trim path))
          (case-fold-search nil))
      (when (string-match "\\`\\(?:.*?/\\)?\\(Sync/.*\\)\\'" trimmed)
        (expand-file-name (match-string 1 trimmed) "~")))))
#+end_src

#+begin_src emacs-lisp :tangle "./boox-core.el"
(defun a3madkour/boox--slug (title &optional max-length)
  "Turn TITLE into a filename-safe slug of at most MAX-LENGTH characters.
MAX-LENGTH defaults to 40.  Characters illegal on Android/FAT are replaced
with hyphens, runs of hyphens are collapsed so the result can never contain
`a3madkour/boox-key-separator', and leading/trailing hyphens are trimmed.
Returns \"untitled\" when nothing usable remains."
  (let* ((max (or max-length 40))
         (s (or title ""))
         (s (replace-regexp-in-string "[/\\\\:*?\"<>|]" "-" s))
         (s (replace-regexp-in-string "[[:space:]]+" "-" (string-trim s)))
         (s (replace-regexp-in-string "-+" "-" s))
         (s (if (> (length s) max) (substring s 0 max) s))
         (s (replace-regexp-in-string "\\`-+\\|-+\\'" "" s)))
    (if (string-empty-p s) "untitled" s)))
#+end_src

#+begin_src emacs-lisp :tangle "./boox-core.el"
(defun a3madkour/boox--key-from-filename (filename)
  "Extract the pushed key from FILENAME, or nil if it carries none.
Files are pushed as KEY--TITLE.EXT.  NeoReader appends suffixes to the
title half on export, which is why only the half before the first
separator is trusted."
  (let ((base (file-name-nondirectory (or filename ""))))
    (when (string-match (concat "\\`\\([^/]+?\\)"
                                (regexp-quote a3madkour/boox-key-separator))
                        base)
      (match-string 1 base))))

(defun a3madkour/boox--book-key (file existing-keys)
  "Return a key for book FILE that does not collide with EXISTING-KEYS.
EXISTING-KEYS should include every key already in use, both book keys and
bibliography citekeys, so a book can never shadow a citekey.  Collisions
get a numeric suffix."
  (let* ((root (a3madkour/boox--slug (file-name-base file) 32))

(defvar a3madkour/boox-export-separator-regexp "^[-=_]\\{5,\\}[ \t]*$"
  "Regexp matching the line NeoReader writes between annotation blocks.
Retune against a real capture; see tests/fixtures/.")

(defvar a3madkour/boox-export-page-regexp "\\(?:[Pp]age\\|P\\.\\)[ \t]*\\([0-9]+\\)"
  "Regexp whose first group captures the page number within a block.")

(defvar a3madkour/boox-export-note-regexp
  "^[ \t]*\\(?:Annotations?\\|Notes?\\|Comment\\)[：:][ \t]*"
  "Regexp matching the label that introduces a typed note within a block.")

(defvar a3madkour/boox-export-text-regexp
  "^[ \t]*\\(?:Original Text\\|Highlight\\|Text\\)[：:][ \t]*"
  "Regexp matching the label that introduces highlighted text within a block.")

(defun a3madkour/boox--parse-block (block)
  "Parse one BLOCK of a NeoReader export into a plist, or nil.
The plist has :page, :text and :note.  A block without a page number or
without highlighted text is discarded, which is what removes the document
header block that precedes the first separator."
  (let ((page (when (string-match a3madkour/boox-export-page-regexp block)
                (string-to-number (match-string 1 block))))
        text note)
    (when page
      (let* ((lines (split-string block "\n"))
             (state 'text)
             (text-lines nil)
             (note-lines nil))
        (dolist (line lines)
          (cond
           ((string-match-p a3madkour/boox-export-note-regexp line)
            (setq state 'note)
            (let ((rest (replace-regexp-in-string
                         a3madkour/boox-export-note-regexp "" line)))
              (unless (string-empty-p (string-trim rest))
                (push rest note-lines))))
           ((string-match-p a3madkour/boox-export-text-regexp line)
            (setq state 'text)
            (let ((rest (replace-regexp-in-string
                         a3madkour/boox-export-text-regexp "" line)))
              (unless (string-empty-p (string-trim rest))
                (push rest text-lines))))
           ((string-match-p a3madkour/boox-export-page-regexp line) nil)
           ((string-empty-p (string-trim line)) nil)
           ((eq state 'note) (push line note-lines))
           (t (push line text-lines))))
        (setq text (string-trim (mapconcat #'identity (nreverse text-lines) " "))
              note (string-trim (mapconcat #'identity (nreverse note-lines) " "))))
      (unless (string-empty-p text)
        (list :page page
              :text text
              :note (unless (string-empty-p note) note))))))

(defun a3madkour/boox--parse-highlights (content)
  "Parse NeoReader export CONTENT into a list of (:page :text :note) plists.
Returns nil when nothing parses, which callers must report rather than
writing an empty highlight section."
  (when (and content (stringp content))
    (delq nil (mapcar #'a3madkour/boox--parse-block
                      (split-string content
                                    a3madkour/boox-export-separator-regexp
                                    t)))))

** Annotated-file recognition
NeoReader's "Embed to PDF" writes annotations *into the pushed file itself* —
same directory, same filename, larger.  There is no separately-named export, so
a file is recognised as annotated by comparing it against the library original
it was pushed from.

Both halves depend on this and getting it wrong destroys work: [[*Push][push]] deletes
reading-directory files it believes are its own stale output, and an annotated
file is byte-different but name-identical to what push put there.
#+begin_src emacs-lisp :tangle "./boox-core.el"
(defun a3madkour/boox--file-sha256 (file)
  "Return the SHA-256 of FILE's contents, or nil when unreadable."
  (when (and file (file-readable-p file))
    (with-temp-buffer
      (set-buffer-multibyte nil)
      (insert-file-contents-literally file)
      (secure-hash 'sha256 (current-buffer)))))

(defun a3madkour/boox--annotated-p (file source)
  "Non-nil when FILE differs from the library SOURCE it was pushed from.
Size is checked first because it settles the common case without reading
either file; the hash only runs when sizes match, which catches an edit
that happens to preserve length."
  (when (and file source (file-readable-p file) (file-readable-p source))
    (let ((sf (file-attribute-size (file-attributes file)))
          (ss (file-attribute-size (file-attributes source))))
      (or (/= sf ss)
          (not (equal (a3madkour/boox--file-sha256 file)
                      (a3madkour/boox--file-sha256 source)))))))

(defun a3madkour/boox--queue-entries ()
  "Return a list of plists describing every :active: entry in queue.org.
Each plist has :key, :source, :title and :kind (either `paper' or `book').
Book entries without a :BOOX_KEY: get one assigned and written back, so the

  (require 'citar)
  (with-current-buffer (find-file-noselect (a3madkour/org-file "queue.org"))
    (let ((known (a3madkour/boox--all-known-keys))
          (dirty nil)
          (entries nil))
      (org-map-entries
       (lambda ()
         (let ((citekey (org-entry-get nil "CITEKEY"))
               (reading-file (org-entry-get nil "READING_FILE")))

            (citekey

               (push (list :key key
                           :kind 'book
                           :source source

       "+active" 'file)

(defun a3madkour/boox-push-queue ()
  "Stage every :active: queue entry into the Boox reading directory.
Files whose queue entry is no longer active are removed, which removes them
from the tablet too.  Only files this command created are ever removed."
  (interactive)
  (let* ((dir (file-name-as-directory a3madkour/boox-reading-directory))
         (entries (a3madkour/boox--queue-entries))
         copied skipped removed missing unsupported)
    (unless (file-directory-p dir) (make-directory dir t))
    (dolist (entry entries)
      (let ((src (plist-get entry :source)))
        (cond
         ((or (null src) (not (file-readable-p src)))
          (push (plist-get entry :key) missing))
         ((not (member (downcase (or (file-name-extension src) ""))
                       a3madkour/boox-supported-extensions))
          (push (cons (plist-get entry :key) (file-name-nondirectory src)) unsupported))
         (t
          (let* ((name (a3madkour/boox--pushed-filename entry))
                 (dest (expand-file-name name dir)))
            (if (file-exists-p dest)
                (push (cons (plist-get entry :key) name) skipped)
              (copy-file src dest)
              (push (cons (plist-get entry :key) name) copied)))))))
    (let ((wanted (mapcar #'a3madkour/boox--pushed-filename
                          (seq-filter (lambda (e) (plist-get e :source)) entries))))
      (dolist (f (a3madkour/boox--pushed-files))
        (unless (member f wanted)
          (delete-file (expand-file-name f dir))
          (push f removed))))
    (with-current-buffer (get-buffer-create "*Boox Sync*")
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert (format "Boox reading directory: %s\n\n" dir))
        (insert (format "Copied: %d   Skipped: %d   Removed: %d   Missing: %d   Unsupported: %d\n\n"
                        (length copied) (length skipped) (length removed)
                        (length missing) (length unsupported)))
        (dolist (pair (list (cons "Copied" copied)
                            (cons "Skipped (already present)" skipped)
                            (cons "Removed (no longer active)" removed)
                            (cons "Missing source" missing)
                            (cons "Unsupported format" unsupported)))
          (when (cdr pair)
            (insert (car pair) ":\n")
            (dolist (it (nreverse (cdr pair)))
              (insert (format "  %s\n" (if (consp it) (format "%s  (%s)" (cdr it) (car it)) it))))
            (insert "\n"))))
      (special-mode))
    (display-buffer "*Boox Sync*")
    (message "Boox: %d copied, %d skipped, %d removed, %d missing, %d unsupported"
             (length copied) (length skipped) (length removed)
             (length missing) (length unsupported))))

              (org-entry-get (point-min) "BOOX_IMPORTED_SHA")))))
    (or (and recorded
             (equal recorded (a3madkour/boox--file-sha256 file)))
        (let ((archived (expand-file-name (file-name-nondirectory file)
                                          a3madkour/boox-archive-directory)))
          (and (file-exists-p archived)
               (= (file-attribute-size (file-attributes archived))
                  (file-attribute-size (file-attributes file))))))))

(defun a3madkour/boox--scan-inbox ()
  "Return a list of candidate plists for every importable file found.
Each plist has :file, :key, :kind (`pdf' or `text'), :status and :selected."
  (let (candidates)
    (dolist (dir a3madkour/boox-inbox-directories)
      (when (file-directory-p dir)
        (dolist (f (directory-files dir t "\\`[^.]" t))
          (let* ((name (file-name-nondirectory f))
                 (ext (downcase (or (file-name-extension f) "")))
                 (key (a3madkour/boox--key-from-filename f)))
            (cond
             ((string-match-p "sync-conflict" name) nil)
             ((file-directory-p f) nil)
             ;; A pushed PDF whose library original cannot be resolved is
             ;; reported rather than skipped: silently dropping it looks
             ;; identical to "nothing to import" and hides the real fault,
             ;; which is usually that citar has not parsed the bibliography
             ;; yet in this session.
             ((and (string= ext "pdf")
                   key
                   (null (a3madkour/boox--reading-file-source f)))
              (push (list :file f :key key :kind 'pdf
                          :status 'source-unresolved :selected nil)
                    candidates))
             ;; A pushed PDF only becomes a candidate once it differs from the
             ;; library original, i.e. once NeoReader has embedded annotations
             ;; into it.  Untouched pushed files are still being read.
             ((and (string= ext "pdf")
                   (not (a3madkour/boox--annotated-p
                         f (a3madkour/boox--reading-file-source f))))
              nil)
             ((not (member ext '("pdf" "txt" "html" "md"))) nil)
             ((a3madkour/boox--already-imported-p f)
              (push (list :file f :key key :kind (if (string= ext "pdf") 'pdf 'text)
                          :status 'already-imported :selected nil)
                    candidates))
             ((null key)
              (push (list :file f :key nil :kind (if (string= ext "pdf") 'pdf 'text)
                          :status 'no-key :selected nil)
                    candidates))
             (t
              (push (list :file f :key key
                          :kind (if (string= ext "pdf") 'pdf 'text)
                          :status 'ready :selected t)
                    candidates)))))))
    (nreverse candidates)))

(defun a3madkour/boox--pdf-annotations (file)
  "Return the user's annotations in FILE as a page-ordered list of plists.
Each plist has :page, :type, :text and :note.  Highlight-family annotations
carry their underlying text, recovered from the PDF's own text layer via the
annotation's markup edges — NeoReader leaves /Contents empty, so the text is
not in the annotation itself.  Ink strokes carry no text and are collapsed to
one entry per page.  The document's own link annotations are ignored."
  (require 'pdf-info)
  (let (marks ink-pages)
    (dolist (a (pdf-info-getannots nil file))
      (let* ((type (cdr (assq 'type a)))
             (page (cdr (assq 'page a)))
             (edges (cdr (assq 'markup-edges a)))
             (contents (cdr (assq 'contents a)))
             (note (unless (string-empty-p (string-trim (or contents "")))
                     (string-trim contents))))
        (cond
         ((memq type '(highlight underline squiggly strike-out))
          (push (list :page page :type type :note note
                      :text (string-trim
                             (replace-regexp-in-string
                              "[ \t\n\r]+" " "
                              (mapconcat (lambda (e) (pdf-info-gettext page e nil file))
                                         edges " "))))
                marks))
         ((eq type 'ink)
          (unless (assq page ink-pages)
            (push (cons page note) ink-pages))))))
    (dolist (p ink-pages)
      (push (list :page (car p) :type 'ink :text nil :note (cdr p)) marks))
    (sort (nreverse marks)
          (lambda (x y) (< (plist-get x :page) (plist-get y :page))))))

(defun a3madkour/boox--annotation-summary (marks)
  "Human-readable count of MARKS, split by kind."
  (let ((h (seq-count (lambda (m) (plist-get m :text)) marks))
        (i (seq-count (lambda (m) (eq (plist-get m :type) 'ink)) marks)))
    (cond ((and (zerop h) (zerop i)) "no annotations found")
          ((zerop i) (format "%d highlight%s" h (if (= h 1) "" "s")))
          ((zerop h) (format "handwriting on %d page%s" i (if (= i 1) "" "s")))
          (t (format "%d highlight%s + handwriting on %d page%s"
                     h (if (= h 1) "" "s") i (if (= i 1) "" "s"))))))

(defun a3madkour/boox--source-for-key (key)
  "Return the original source file for KEY, paper or book."
  (or (when-let ((entry (ignore-errors (citar-get-entry key))))
        (when-let* ((file (alist-get "file" entry nil nil #'string=))
                    (raw (car (seq-filter
                               (lambda (p) (string-match-p "\\.\\(pdf\\|epub\\)\\'"
                                                           (string-trim p)))
                               (split-string file ";")))))
          (or (a3madkour/normalize-library-path raw) (string-trim raw))))
      (plist-get (seq-find (lambda (e) (equal (plist-get e :key) key))
                           (a3madkour/boox--queue-entries))
                 :source)))

(defun a3madkour/boox--ensure-book-note (key source)
  "Return the ref-note file for book KEY, creating it against SOURCE if absent."
  (let ((file (expand-file-name (concat key ".org") org-ref-notes)))
    (unless (file-exists-p file)
      (with-current-buffer (find-file-noselect file)
        (insert (format ":PROPERTIES:\n:ID: %s\n:NOTER_DOCUMENT: %s\n:ORIGINAL_DOCUMENT: %s\n:DATE_CREATED: %s\n:LAST_MODIFIED: %s\n:END:\n#+title: %s\n#+filetags: :Ref:Book:\n\n"
                        (org-id-new)
                        source source
                        (format-time-string "[%Y-%m-%d %a]")
                        (format-time-string "[%Y-%m-%d %a]")
                        (file-name-base source)))
        (save-buffer)))
    file))

(defun a3madkour/boox--note-for-key (key)
  "Return the ref-note file for KEY, creating it if necessary."
  (let ((citar-note (expand-file-name (concat key ".org") org-ref-notes)))
    (cond
     ((file-exists-p citar-note) citar-note)
     ((ignore-errors (citar-get-entry key))
      (citar-create-note key)

     (t (a3madkour/boox--ensure-book-note key (a3madkour/boox--source-for-key key))))))

(defun a3madkour/boox--annotated-destination (source)
  "Path the annotated copy of SOURCE should occupy."
  (expand-file-name
   (format "%s%s.%s"
           (file-name-base source)
           a3madkour/boox-annotated-suffix
           (file-name-extension source))
   (file-name-directory source)))

(defun a3madkour/boox--file-annotated-pdf (key file)
  "Copy annotated FILE into the library beside KEY's original.
Returns the destination path, or nil when the source cannot be resolved.
Deliberately writes no properties: creating the ref note can rewrite it
wholesale (`a3madkour/citar-org-format-note' erases the buffer), so every
property is written in one final pass by `a3madkour/boox--record-documents'
after the note is settled."
  (when-let* ((source (a3madkour/boox--source-for-key key))
              (dest (a3madkour/boox--annotated-destination source)))
    (when (or (not (file-exists-p dest))
              (y-or-n-p (format "Overwrite existing %s? "
                                (file-name-nondirectory dest))))
      (copy-file file dest t)
      dest)))

(defun a3madkour/boox--record-documents (key source dest sha)
  "Point KEY's ref note at DEST, remembering SOURCE and SHA.
Run last, so nothing downstream can rewrite the note over these values.
Signals a `user-error' if the properties do not survive the save, rather
than reporting a success that did not happen."
  (let ((note (a3madkour/boox--note-for-key key)))
    (with-current-buffer (find-file-noselect note)
      (org-entry-put (point-min) "ORIGINAL_DOCUMENT" source)
      (org-entry-put (point-min) "ANNOTATED_DOCUMENT" dest)
      (org-entry-put (point-min) "NOTER_DOCUMENT" dest)
      (org-entry-put (point-min) "BOOX_IMPORTED_SHA" sha)
      (org-entry-put (point-min) "LAST_MODIFIED" (format-time-string "[%Y-%m-%d %a]"))
      (save-buffer))
    ;; Re-read from disk: a stale buffer elsewhere can silently win the write.
    (with-temp-buffer
      (insert-file-contents note)
      (goto-char (point-min))
      (unless (search-forward (concat ":NOTER_DOCUMENT: " dest) nil t)
        (user-error "Boox: wrote %s but NOTER_DOCUMENT did not stick in %s"
                    (file-name-nondirectory dest) (file-name-nondirectory note))))
    note))

(defun a3madkour/boox--insert-highlights (key highlights)
  "Insert HIGHLIGHTS into KEY's ref note under a `Tablet Highlights' heading.
Existing content under that heading is replaced, so re-importing after more
reading supersedes the earlier pass instead of duplicating it.  Entries
without text are handwriting: they get a heading and a `NOTER_PAGE' so they
are still clickable, but no quoted body."
  (with-current-buffer (find-file-noselect (a3madkour/boox--note-for-key key))
    (org-with-wide-buffer
     (goto-char (point-min))
     (if (re-search-forward "^\\* Tablet Highlights[ \t]*$" nil t)
         (let ((start (line-beginning-position)))
           (org-end-of-subtree t t)
           (delete-region start (point))
           (goto-char start))
       (goto-char (point-max))
       (unless (bolp) (insert "\n")))
     (insert "* Tablet Highlights\n")
     (dolist (h highlights)
       (let ((text (plist-get h :text))
             (page (plist-get h :page)))
         (insert (format "** %s\n"
                         (if text
                             (truncate-string-to-width text 70 nil nil "…")
                           (format "Handwritten note — page %s" page))))
         (when page
           (insert (format ":PROPERTIES:\n:NOTER_PAGE: %d\n:END:\n" page)))
         (when text (insert (format "%s\n" text)))
         (when (plist-get h :note)
           (insert (format "\n%s\n" (plist-get h :note))))
         (insert "\n")))
     (org-entry-put (point-min) "LAST_MODIFIED" (format-time-string "[%Y-%m-%d %a]"))
     (save-buffer))))

(defun a3madkour/boox-import-execute ()
  "File every selected candidate and record what was imported.
Annotated PDFs are copied into the library beside their original but are
deliberately left in the reading directory: that file is the live copy on
the tablet, and moving it would delete the paper mid-read.  Re-import is
prevented instead by recording the imported content hash on the ref note,
so embedding more annotations later makes the file importable again."
  (interactive)
  (unless (file-directory-p a3madkour/boox-archive-directory)
    (make-directory a3madkour/boox-archive-directory t))
  (let ((selected (seq-filter (lambda (c) (plist-get c :selected))
                              a3madkour/boox--candidates))
        (results nil))
    (dolist (c selected)
      (let ((key (plist-get c :key))
            (file (plist-get c :file)))
        (pcase (plist-get c :kind)
          ('pdf
           (if-let ((dest (a3madkour/boox--file-annotated-pdf key file)))
               (let ((marks (a3madkour/boox--pdf-annotations file)))
                 ;; Highlights first: inserting them may create the ref note,
                 ;; and note creation rewrites the file wholesale.  Properties
                 ;; go last so nothing can clobber them.
                 (when marks (a3madkour/boox--insert-highlights key marks))
                 (a3madkour/boox--record-documents
                  key (a3madkour/boox--source-for-key key) dest
                  (a3madkour/boox--file-sha256 file))
                 (push (format "%s → %s, %s" key (abbreviate-file-name dest)
                               (a3madkour/boox--annotation-summary marks))
                       results))
             (push (format "%s → SKIPPED, source unresolved" key) results)))
          ('text
           (let ((highlights (a3madkour/boox--parse-highlights
                              (a3madkour/boox--file-contents file))))
             (if (null highlights)
                 (push (format "%s → 0 highlights parsed, NOT imported" key) results)
               (a3madkour/boox--insert-highlights key highlights)
               (push (format "%s → %d highlights" key (length highlights)) results)
               (rename-file file (expand-file-name
                                  (file-name-nondirectory file)
                                  a3madkour/boox-archive-directory)
                            t)))))))
    (message "Boox import: %s" (string-join (nreverse results) "; "))
    (a3madkour/boox-import)))

(defun a3madkour/-project-heading-string (name area status done-enough goal repo include-placeholder)
  "Build the heading + property drawer string for a new project.
GOAL is used as-is, or replaced with \"exploratory\" if empty.
REPO is a directory path (string) or nil. INCLUDE-PLACEHOLDER appends an
empty `** TODO ' line — useful when the caller wants the user to fill it
in interactively, omitted when the heading will be filled by refile."
  (format (concat "* %s"
                  " :%s:%s:project:\n"
                  ":PROPERTIES:\n"
                  ":TRIGGER: next-sibling todo!(NEXT)\n"
                  ":ORDERED: t\n"
                  ":DONE_ENOUGH: %s\n"
                  ":GOAL: %s\n"
                  "%s"
                  ":END:\n\n"
                  "%s")
          name area status
          done-enough
          (if (string-empty-p goal) "exploratory" goal)
          (if repo
              (format ":REPO: %s\n" (abbreviate-file-name repo))
            "")
          (if include-placeholder "** TODO \n" "")))

(defun a3madkour/create-project ()
  "Create a new project heading in projects.org with required properties.
When called from code, returns a marker to the new project heading.
When called interactively, also switches to the projects buffer and
inserts a placeholder \"** TODO \" line ready for editing."
  (interactive)
  (let* ((interactive-call (called-interactively-p 'any))
         (name (read-string "Project name: "))
         (done-enough (read-string "Done enough (what does finished look like?): "))
         (goal (read-string "Which goal does this serve? (blank if exploratory): "))
         (area (completing-read "Life area: "
                                '("artistic" "intellectual" "health" "career" "social")
                                nil t))
         (status (completing-read "Status: " '("active" "inactive") nil t))
         (has-repo (yes-or-no-p "Does this project have a git repo? "))
         (repo (when has-repo (read-directory-name "Repo path: "))))
    (when (string= status "active")
      (let ((count (a3madkour/active-project-count)))
        (when (>= count a3madkour/project-wip-limit)
          (user-error "WIP limit reached (%d/%d active). Deactivate one first"
                      count a3madkour/project-wip-limit))))
    (let ((heading (a3madkour/-project-heading-string
                    name area status done-enough goal repo interactive-call))
		  project-marker)
	  (with-current-buffer (find-file-noselect (a3madkour/org-file "projects.org"))
        (goto-char (point-max))
        (insert heading)
        (save-buffer)

	  (when interactive-call
		(switch-to-buffer (find-file-noselect (a3madkour/org-file "projects.org")))
        (goto-char (marker-position project-marker))
        (re-search-forward "^\\*\\* TODO ")
        (end-of-line))

	  project-marker)))

(defun a3madkour/create-project-notes ()
  "Create an org-roam note for the project at point and link it."
  (interactive)
  (unless (member "project" (org-get-tags))
    (user-error "Not on a project heading"))
  (let* ((title (org-get-heading t t t t))
         (note-title (read-string "Note title: " title ))
         (slug (org-roam-node-slug (org-roam-node-create :title note-title)))
         (dir (expand-file-name "projects" org-roam-directory))
         (filepath (expand-file-name (concat slug ".org") dir))
         (id nil)
         (has-context nil))
    (make-directory dir t)
    ;; Create the file if it doesn't exist, otherwise extract its ID
    (if (file-exists-p filepath)
        (with-current-buffer (find-file-noselect filepath)
		  (goto-char (point-min))
		  (when (re-search-forward "^:ID:\\s-+\\(.+\\)$" nil t)
            (setq id (match-string 1))))
	  (setq id (org-id-new))
	  (with-current-buffer (find-file-noselect filepath)
        (insert (format ":PROPERTIES:\n:ID: %s\n:END:\n#+title: %s\n#+filetags: :project:\n\n* Context\n\n* Decisions\n\n* References\n"
                        id note-title))
        (save-buffer))
	  (org-roam-db-update-file filepath))
    ;; Add context link if not already present
    (when id
	  (save-excursion
        (org-end-of-meta-data t)
        (let ((bound (save-excursion (org-end-of-subtree t t) (point))))
		  (setq has-context (re-search-forward "^Context: \\[\\[id:" bound t))))
	  (unless has-context
        (org-end-of-meta-data t)
        (insert (format "Context: [[id:%s][%s]]\n" id note-title))
        (save-buffer)))
    (find-file filepath)
    (message "Opened: %s" note-title)))

  "Marker to the inbox subtree being sent to org-roam, removed on capture finalize.")

(defun a3madkour--inbox-to-roam-finalize ()
  "Hook: cut the pending inbox subtree if the org-roam capture finalized successfully."
  (unwind-protect
      (when (and (not org-note-abort)
                 (markerp a3madkour--inbox-to-roam-marker)
                 (marker-buffer a3madkour--inbox-to-roam-marker))
        (with-current-buffer (marker-buffer a3madkour--inbox-to-roam-marker)

(defun a3madkour/inbox-to-org-roam ()

Preserves the inbox item's body as initial content of the new node.
The inbox subtree is removed only after the capture finalizes successfully;
on abort, the inbox item stays where it was."
  (org-back-to-heading t)
  (let* ((title (org-get-heading t t t t))

                 (org-end-of-meta-data t)

                  (buffer-substring-no-properties

                   (org-entry-end-position))))))

(defun a3madkour/-build-habit-repeater (frequency flexibility)
  "Compute the org-mode repeater string for a habit.
FREQUENCY is one of \"daily\", \"every 2 days\", \"every 3 days\",
\"weekly\", \"biweekly\", \"monthly\". FLEXIBILITY (a string) appends a
`/Nd' tolerance window but only for the daily/every-N-days frequencies."
  (let ((base (cond
               ((string= frequency "daily") ".+1d")
               ((string= frequency "every 2 days") ".+2d")
               ((string= frequency "every 3 days") ".+3d")
               ((string= frequency "weekly") ".+1w")
               ((string= frequency "biweekly") ".+2w")
               ((string= frequency "monthly") ".+1m"))))
    (if (and flexibility
             (not (string-empty-p flexibility))
             (member frequency '("daily" "every 2 days" "every 3 days")))
        (format "%s/%sd" base (string-to-number flexibility))
      base)))

                        (read-string "Max days before overdue (blank for same): ")))

(defun a3madkour/process-inbox-item ()
  "Process the current inbox item: classify, set properties, and refile."
  (interactive)
  (unless (org-entry-get nil "TODO")
    (user-error "Not on a TODO heading"))
  (let ((destination (completing-read "Where does this go? "
                                      '("action" "project" "someday" "org-roam" "delete")
                                      nil t)))
    (cond
     ((string= destination "delete")
      (a3madkour/inbox-delete))
     ((string= destination "org-roam")
      (a3madkour/inbox-to-org-roam))
     ((string= destination "someday")

                                             '("existing project" "create new project")
                                             nil t)))

            (a3madkour/inbox-to-new-project)
          (a3madkour/inbox-to-existing-project))))
     (t (user-error "Unknown destination: %s" destination)))))
